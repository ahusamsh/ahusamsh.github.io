function toggleRoutine(id){const item=state.routines.find(x=>x.id===id);if(!item)return;item.done=!item.done;saveLocal();render();toast(item.done?"أحسنت يا بطل! ⭐":"تم التعديل")}
function togglePrayer(name){const item=state.prayers.find(x=>x.name===name);if(!item)return;if(!item.done){item.done=true;item.onTime=false}else if(!item.onTime)item.onTime=true;else{item.done=false;item.onTime=false}saveLocal();render();toast(item.done?(item.onTime?"ما شاء الله، في وقتها! 🌟":"تقبل الله 🕌"):"تم التعديل")}
function changeQuran(id,stage){const item=state.quran.find(x=>x.id===id);if(!item)return;item.stage=stage;saveLocal();render();toast("تم تحديث رحلة الحفظ 📖")}
function redeemReward(id){const reward=state.rewards.find(x=>x.id===id);if(!reward||state.points<reward.cost)return;state.redemptions.unshift({id:newId(),rewardId:id,cost:reward.cost,status:"pending",requestedAt:new Date().toISOString()});saveLocal();render();toast("تم إرسال الطلب للوالد 🎁")}
function deleteRoutine(id){state.routines=state.routines.filter(x=>x.id!==id);saveLocal();render()}
function deleteReward(id){state.rewards=state.rewards.filter(x=>x.id!==id);saveLocal();render()}
function updateRedemption(id,status){const red=state.redemptions.find(x=>x.id===id);if(!red)return;red.status=status;red.decidedAt=new Date().toISOString();saveLocal();render();toast(status==="approved"?"تمت الموافقة ✅":"تم رفض الطلب")}
function resetDay(){state.routines.forEach(x=>x.done=false);state.prayers.forEach(x=>{x.done=false;x.onTime=false});saveLocal();render();toast("تمت إعادة ضبط اليوم")}
