function scriptRequest(payload){
  return new Promise((resolve,reject)=>{
    const requestId=newId();
    const iframe=document.createElement("iframe");
    iframe.hidden=true;
    iframe.name=`hero-api-${requestId}`;
    const cleanup=()=>{clearTimeout(timer);window.removeEventListener("message",onMessage);iframe.remove()};
    const onMessage=event=>{
      const data=event.data;
      if(event.source!==iframe.contentWindow||!data||data.source!=="little-hero-api"||data.requestId!==requestId)return;
      cleanup();
      if(!data.ok)reject(new Error(data.error||"فشلت المزامنة"));else resolve(data);
    };
    const timer=setTimeout(()=>{cleanup();reject(new Error("تعذر الوصول إلى Google Sheet"))},20000);
    window.addEventListener("message",onMessage);
    document.body.appendChild(iframe);
    if(payload.action==="load"){
      const separator=apiUrl().includes("?")?"&":"?";
      iframe.src=`${apiUrl()}${separator}mode=load&requestId=${encodeURIComponent(requestId)}`;
      return;
    }
    const form=document.createElement("form");
    form.hidden=true;
    form.method="POST";
    form.action=apiUrl();
    form.target=iframe.name;
    const input=document.createElement("input");
    input.name="payload";
    input.value=JSON.stringify({...payload,requestId});
    form.appendChild(input);
    document.body.appendChild(form);
    form.submit();
    form.remove();
  });
}
