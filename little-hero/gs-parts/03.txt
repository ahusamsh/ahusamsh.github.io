function ensureStructure_(ss) {
  Object.values(SHEETS).forEach(name=>{ if (!ss.getSheetByName(name)) ss.insertSheet(name); });
  const settings = ss.getSheetByName(SHEETS.settings);
  if (settings.getLastRow() === 0) {
    settings.getRange(1,1,1,3).setValues([['المفتاح','القيمة','الوصف']]);
    formatHeader_(settings, 3);
  }
  initializeHeaders_(ss);
  refreshDashboard_(ss);
}

function initializeHeaders_(ss) {
  const definitions = {};
  definitions[SHEETS.children] = ['id','الاسم','نشط','تاريخ الإنشاء','آخر تحديث'];

  definitions[SHEETS.prayers] = ['id','child_id','التاريخ','الصلاة','مكتملة','في_وقتها','النقاط','وقت الإنجاز','الجهاز','آخر تحديث'];
  definitions[SHEETS.quran] = ['id','child_id','رقم_السورة','اسم_السورة','المرحلة','النقاط','الترتيب','آخر مراجعة','آخر تحديث'];
  definitions[SHEETS.rewards] = ['id','child_id','الرمز','المكافأة','التكلفة','نشط','تاريخ الإنشاء','آخر تحديث'];
  definitions[SHEETS.redemptions] = ['id','child_id','reward_id','التكلفة','الحالة','وقت الطلب','وقت القرار','آخر تحديث'];
  definitions[SHEETS.points] = ['id','child_id','المبلغ','النوع','المصدر','source_id','السبب','التاريخ','وقت الإنشاء'];
  definitions[SHEETS.events] = ['id','child_id','العملية','التفاصيل_json','الجهاز','وقت الإنشاء'];

    const sh = ss.getSheetByName(name);
